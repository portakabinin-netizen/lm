          source.expoPublicEnv && typeof source.expoPublicEnv === 'object'

          if (key.startsWith('EXPO_PUBLIC_') && typeof val === 'string' && val.trim()) {

      candidates.map(async (url) => {
        try {
          const res = await axios.get(`${url}auth/health-check`, { timeout: 5000 });
          const isSuspended =
            typeof res.data === 'string' &&
            (res.data.includes('<title>Service Suspended</title>') || res.data.includes('<!DOCTYPE html>'));
          const isDbDown = res.data?.database === 'disconnected';

          if (res.status === 200 && !isSuspended && !isDbDown) {
            liveSet.add(url);
          }
        } catch {
          // Host unreachable or timed out
        }
      })
    );

    this.liveURLs = Array.from(liveSet);

      `[AppService] Checked ${candidates.length} candidate hosts: ${this.liveURLs.length} live host(s) found`,
      this.liveURLs
    );
    return [...this.liveURLs];
